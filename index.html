      Optional Firebase backup setup
      Enable Anonymous Authentication and Cloud Firestore in your Firebase project. Add this site to authorized domains if required. Use owner-only Firestore rules, for example:
      rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /practiceRoomBackups/{uid} {
      allow read, write: if request.auth != null
                         && request.auth.uid == uid;
    }
  }
}
